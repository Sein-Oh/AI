# doc-classifier 런타임 의존성 (버전은 루트 constraints.txt 기준, 모두 표준 WinPython 환경에 포함)
# 설치: python -m pip install -r CLI/doc-classifier/requirements.txt -c constraints.txt
# scan / classify / list는 표준 라이브러리만 사용. 아래는 peek(표본 확인)와 엑셀 보고서용 (없으면 JSON 오류로 안내)
pymupdf        # PDF 앞부분 읽기 (peek)
olefile        # HWP 5.0 미리보기 텍스트, 옛 Office(doc/xls/ppt/msg) 본문 (peek)
openpyxl       # xlsx 앞부분 읽기 (peek), 엑셀 보고서 (summary --save)
# 테스트 전용
python-docx    # 테스트 샘플 docx 만들기 (peek는 docx를 표준 zipfile로 읽음)
python-pptx    # 테스트 샘플 pptx 만들기 (peek는 pptx를 표준 zipfile로 읽음)
pywin32        # Windows 실기 테스트 러너 (잠긴 파일 시험)
